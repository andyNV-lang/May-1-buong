const DB = {};           // ten sheet -> mảng 2 chiều (gồm cả hàng tiêu đề)
function mkSheet(name, header) { DB[name] = [header.slice()]; }

class Range {
  constructor(sh, r, c, nr, nc) { Object.assign(this, { sh, r, c, nr, nc }); }
  getValues() {
    const out = [];
    for (let i = 0; i < this.nr; i++) {
      const row = DB[this.sh][this.r - 1 + i] || [];
      const line = [];
      for (let j = 0; j < this.nc; j++) line.push(row[this.c - 1 + j] === undefined ? '' : row[this.c - 1 + j]);
      out.push(line);
    }
    return out;
  }
  setValues(v) {
    // Google Sheets thật NÉM LỖI khi dữ liệu lệch chiều với vùng ghi.
    // Bộ giả lập trước đây ghi bừa nên đã bỏ lọt lỗi dữ liệu mẫu thiếu cột kl_vao
    // (menu "Tạo dữ liệu mẫu" hỏng từ bản 1.3 mà 176 kiểm thử không hề biết).
    if (!Array.isArray(v)) {
      throw new Error('setValues: dữ liệu phải là mảng 2 chiều.');
    }
    if (v.length !== this.nr) {
      throw new Error('The number of rows in the data does not match the number of rows in the range. '
        + 'The data has ' + v.length + ' but the range has ' + this.nr + '. '
        + '[sheet ' + this.sh + ']');
    }
    for (let q = 0; q < v.length; q++) {
      if (!Array.isArray(v[q])) {
        throw new Error('setValues: dòng ' + (q + 1) + ' không phải mảng. [sheet ' + this.sh + ']');
      }
      if (v[q].length !== this.nc) {
        throw new Error('The number of columns in the data does not match the number of columns in the range. '
          + 'The data has ' + v[q].length + ' but the range has ' + this.nc + '. '
          + '[sheet ' + this.sh + ', dòng dữ liệu thứ ' + (q + 1) + ']');
      }
    }
    for (let i = 0; i < v.length; i++) {
      const idx = this.r - 1 + i;
      while (DB[this.sh].length <= idx) DB[this.sh].push([]);
      for (let j = 0; j < v[i].length; j++) DB[this.sh][idx][this.c - 1 + j] = v[i][j];
    }
    return this;
  }
  clearContent() {
    for (let i = 0; i < this.nr; i++) {
      const idx = this.r - 1 + i;
      if (DB[this.sh][idx]) for (let j = 0; j < this.nc; j++) DB[this.sh][idx][this.c - 1 + j] = '';
    }
    return this;
  }
  setFontWeight() { return this; } setBackground() { return this; }
  setFontColor() { return this; } setVerticalAlignment() { return this; }
  /* Ghi lại định dạng số để kiểm thử soát được: trước bản 1.6 bộ giả lập KHÔNG có
     hai hàm này, nên cả khối định dạng cột trong 02_Setup.gs (nằm trong try/catch)
     ném lỗi rồi bị nuốt — tức kiểm thử chưa từng chạy qua nó lấy một lần. */
  setNumberFormat(f) {
    for (let i = 0; i < this.nr; i++) {
      for (let j = 0; j < this.nc; j++) DINH_DANG[this.sh + '!' + (this.r + i) + ',' + (this.c + j)] = f;
    }
    return this;
  }
}
const DINH_DANG = {};

class Sheet {
  constructor(name) { this.name = name; }
  getRange(r, c, nr, nc) { return new Range(this.name, r, c, nr || 1, nc || 1); }
  getLastRow() { return DB[this.name].length; }
  getLastColumn() { return DB[this.name][0].length; }
  appendRow(row) { DB[this.name].push(row.slice()); }
  deleteRow(n) { DB[this.name].splice(n - 1, 1); }
  setFrozenRows() {} setRowHeight() {} setColumnWidth() {}
  getMaxRows() { return Math.max(1000, DB[this.name].length); }
  getProtections() { return []; }
  protect() { return { setDescription() { return this; }, setWarningOnly() { return this; } }; }
}
